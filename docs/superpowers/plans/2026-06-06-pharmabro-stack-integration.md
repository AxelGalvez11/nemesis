# PharmaBro Stack Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Integrate 13 curated libraries and tools across five phases to harden PharmaBro's UI, reliability, observability, data pipeline, and platform distribution surface.

**Architecture:** Each phase is independently deployable and testable. Phases 1-2 have no inter-phase dependencies and can run in parallel. Phase 3 (observability) follows Phase 2 (Sentry must be wired first). Phases 4-5 are independent of Phases 1-3 and can be tackled in any sprint order.

**Tech Stack:** Next.js 16 App Router (`@pharmaorb/web`), Deno Supabase Edge Functions, pnpm monorepo + Turbo, Supabase cloud (`qyjmivntajbigjswhahb`), OpenAI (gpt-4o-mini + gpt-4.1-mini), pgvector, Stripe, Vercel.

**Package names:** web = `@pharmaorb/web`, shared = `@nemesis/shared`, mobile = `@pharmaorb/mobile`.

**Hard rules (apply throughout):**
- Never deploy edge functions with local Docker — always `supabase functions deploy --use-api`
- No tool may bypass the `/ask` edge function pipeline (preScreen → classify → retrieve → generate → citation_enforce → consumeAskQuota)
- RHF, Sonner, nuqs, MCP handler = `apps/web` only. Never import in Deno edge functions
- Zod in edge functions: import via `npm:zod` specifier (Deno npm support, already enabled)

- [ ] **Step 6.3: Create global setup (seed test user)**

Create `apps/web/e2e/global-setup.ts`:

```ts
import { mkdir, writeFile } from 'fs/promises'
import { chromium } from '@playwright/test'

const BASE_URL = process.env.PLAYWRIGHT_BASE_URL ?? 'http://localhost:3000'
const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!
const SERVICE_ROLE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY!
const TEST_EMAIL = 'playwright-smoke@test.pharmabro.internal'
const TEST_PASSWORD = 'Smoke!Test!2026'

export default async function globalSetup() {
  // Seed confirmed test user via admin API (bypasses email confirmation)
  const res = await fetch(`${SUPABASE_URL}/auth/v1/admin/users`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${SERVICE_ROLE_KEY}`,
      apikey: SERVICE_ROLE_KEY,
    },
    body: JSON.stringify({
      email: TEST_EMAIL,
      password: TEST_PASSWORD,
      email_confirm: true,
    }),
  })

  if (!res.ok && res.status !== 422) {
    // 422 = user already exists, which is fine
    const body = await res.text()
    throw new Error(`Failed to seed test user: ${res.status} ${body}`)
  }

  // Sign in through the real UI and save auth state
  const browser = await chromium.launch()
  const context = await browser.newContext()
  const page = await context.newPage()

  await page.goto(`${BASE_URL}/sign-in`)
  await page.fill('input[type="email"]', TEST_EMAIL)
  await page.fill('input[type="password"]', TEST_PASSWORD)
  await page.click('button[type="submit"]')
  await page.waitForURL('**/app**', { timeout: 30_000 })

  await mkdir('.auth', { recursive: true })
  await context.storageState({ path: '.auth/seed.json' })
  await browser.close()
```

```ts
#!/usr/bin/env -S deno run --allow-net --allow-read --allow-env

import { extractText, getDocumentProxy } from 'npm:unpdf'
import pLimit from 'npm:p-limit'
import manifest from '../supabase/functions/core-source-sync/providers/drugs-fda.manifest.json' assert { type: 'json' }

const SUPABASE_URL = Deno.env.get('NEXT_PUBLIC_SUPABASE_URL')!
const SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
const CORE_SYNC_URL = `${SUPABASE_URL}/functions/v1/core-source-sync`

const args = Deno.args
const DRY_RUN = args.includes('--dry-run')
const ONLY_NDA = args.find(a => a.startsWith('--only-nda='))?.split('=')[1]
const BATCH_SIZE = 5
const CONCURRENCY = 3

function cleanPdfText(raw: string): string {
  return raw
    .replace(/\f/g, '\n')
    .replace(/\r\n/g, '\n')
    .replace(/[ \t]{3,}/g, '  ')
    .replace(/\n{4,}/g, '\n\n\n')
    .trim()
}

async function fetchPdf(url: string): Promise<Uint8Array> {
  const res = await fetch(url, {
    headers: { 'User-Agent': 'PharmaBro-Ingest/1.0 (research; contact@nemesis.app)' },
  })
  if (!res.ok) throw new Error(`PDF fetch failed: ${res.status} ${url}`)
  const buf = await res.arrayBuffer()
  return new Uint8Array(buf)
}

async function parsePdf(bytes: Uint8Array): Promise<string> {
  const pdf = await getDocumentProxy(bytes)
  const { text } = await extractText(pdf, { mergePages: true })
  return cleanPdfText(text)
}

async function postBatch(parsedDocs: unknown[]) {
  if (DRY_RUN) {
    console.log(`[dry-run] would POST ${parsedDocs.length} docs`)
    return
  }
  const res = await fetch(CORE_SYNC_URL, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${SERVICE_KEY}`,
    },
    body: JSON.stringify({
      provider: 'drugs_fda',
      opts: { pre_parsed_docs: parsedDocs },
    }),
  })
  if (!res.ok) {
    const body = await res.text()
    throw new Error(`POST failed: ${res.status} ${body}`)
  }
  console.log(`✓ posted ${parsedDocs.length} docs`)
}

const limit = pLimit(CONCURRENCY)

const entries = ONLY_NDA
  ? manifest.filter((e: { nda_number: string }) => e.nda_number === ONLY_NDA)
  : manifest

console.log(`Processing ${entries.length} PDFs (dry-run=${DRY_RUN}, concurrency=${CONCURRENCY})`)

const parsedDocs: unknown[] = []

await Promise.all(
  entries.map((entry: { pdf_url: string; drug_name: string; nda_number: string; review_type: string }) =>
    limit(async () => {
      console.log(`Fetching ${entry.drug_name} (${entry.nda_number})...`)
      try {
        const bytes = await fetchPdf(entry.pdf_url)
        const text = await parsePdf(bytes)
        parsedDocs.push({
          nda_number: entry.nda_number,
          drug_name: entry.drug_name,
          review_type: entry.review_type,
          text,
          char_count: text.length,
        })
        console.log(`✓ ${entry.drug_name}: ${text.length.toLocaleString()} chars`)

        // Post in batches
        if (parsedDocs.length >= BATCH_SIZE) {
          const batch = parsedDocs.splice(0, BATCH_SIZE)
          await postBatch(batch)
          await new Promise(r => setTimeout(r, 500)) // inter-batch delay
        }
      } catch (e) {
        console.error(`✗ ${entry.drug_name}:`, e instanceof Error ? e.message : e)
      }
    })
  )
)

// Post remaining
if (parsedDocs.length > 0) {
  await postBatch(parsedDocs)
}

console.log('Done.')
```

```ts
import { serve } from 'https://deno.land/std@0.208.0/http/server.ts'
import { createClient } from 'npm:@supabase/supabase-js'

const supabase = createClient(
  Deno.env.get('SUPABASE_URL')!,
  Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
)

const CORE_SYNC_URL = `${Deno.env.get('SUPABASE_URL')}/functions/v1/core-source-sync`
const MAX_RETRIES = 3

serve(async () => {
  // Read 10 messages with 30s visibility timeout
  const { data: messages } = await supabase.rpc('pgmq_read', {
    queue_name: 'drug_ingest',
    vt: 30,
    qty: 10,
  })

  if (!messages || messages.length === 0) {
    return new Response(JSON.stringify({ processed: 0 }), { status: 200 })
  }

  let processed = 0
  for (const msg of messages) {
    const job = msg.message as { drug_name: string; nda_number: string; provider: string; retries?: number }
    const retries = job.retries ?? 0

    try {
      const res = await fetch(CORE_SYNC_URL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')}`,
        },
        body: JSON.stringify({ provider: job.provider, drug_name: job.drug_name }),
      })

      if (!res.ok) throw new Error(`core-source-sync returned ${res.status}`)

      // Success: delete from queue
      await supabase.rpc('pgmq_delete', { queue_name: 'drug_ingest', msg_id: msg.msg_id })
      processed++
    } catch (e) {
      console.error(`Failed job ${job.drug_name}/${job.provider}:`, e)
      if (retries >= MAX_RETRIES) {
        // Move to DLQ
        await supabase.rpc('pgmq_send', {
          queue_name: 'drug_ingest_dlq',
          msg: { ...job, failed_at: new Date().toISOString(), error: String(e) },
        })
        await supabase.rpc('pgmq_delete', { queue_name: 'drug_ingest', msg_id: msg.msg_id })
      } else {
        // Update retry count and let visibility timeout re-expose it
        await supabase.rpc('pgmq_send', {
          queue_name: 'drug_ingest',
          msg: { ...job, retries: retries + 1 },
        })
        await supabase.rpc('pgmq_delete', { queue_name: 'drug_ingest', msg_id: msg.msg_id })
      }
    }
  }

  return new Response(JSON.stringify({ processed }), { status: 200 })
})
```

```ts
import { streamText } from 'ai'
import { openai } from '@ai-sdk/openai'
import { NextRequest } from 'next/server'
import { createClient } from '@supabase/supabase-js'

export const runtime = 'nodejs'
export const maxDuration = 60

export async function POST(req: NextRequest) {
  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    { global: { headers: { Authorization: req.headers.get('authorization') ?? '' } } }
  )
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return new Response('Unauthorized', { status: 401 })

  const body = await req.json()

  // Call the Deno /ask edge function first (enforces safety + quota + retrieval)
  const edgeRes = await fetch(`${process.env.NEXT_PUBLIC_SUPABASE_URL}/functions/v1/ask`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: req.headers.get('authorization') ?? '',
    },
    body: JSON.stringify(body),
  })

  if (!edgeRes.ok) {
    const err = await edgeRes.text()
    return new Response(err, { status: edgeRes.status })
  }

  const askResponse = await edgeRes.json()

  // Stream the plain_english_summary as tokens using AI SDK
  const result = streamText({
    model: openai('gpt-4.1-mini'),
    messages: [
      {
        role: 'user',
        content: `Present this drug information response clearly:\n\n${askResponse.plain_english_summary}`,
      },
    ],
    onFinish: () => {
      // Citations and structured fields are appended as a final data chunk
    },
  })

  // Append structured fields after stream
  return result.toDataStreamResponse({
    headers: {
      'X-Ask-Citations': JSON.stringify(askResponse.citations ?? []),
      'X-Ask-Evidence-Grade': askResponse.evidence_grade ?? '',
      'X-Ask-Safety-Flags': JSON.stringify(askResponse.safety_flags ?? []),
      'X-Ask-Refused': String(askResponse.refused_unsupported ?? false),
    },
  })
}
```

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import requests
import tempfile
import os
from docling.document_converter import DocumentConverter

app = FastAPI()
converter = DocumentConverter()

class PdfRequest(BaseModel):
    url: str

@app.post("/parse")
def parse_pdf(req: PdfRequest):
    with tempfile.NamedTemporaryFile(suffix=".pdf", delete=False) as tmp:
        try:
            resp = requests.get(req.url, timeout=120, headers={
                "User-Agent": "PharmaBro-Docling/1.0"
            })
            resp.raise_for_status()
            tmp.write(resp.content)
            tmp.flush()

            result = converter.convert(tmp.name)
            text = result.document.export_to_markdown()
            return {"text": text, "char_count": len(text)}
        finally:
            os.unlink(tmp.name)
```
